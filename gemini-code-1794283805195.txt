        function updateClock() {

        // Sistema de navegación por pestañas
        function switchTab(tabName) {
            const tabs = ['cells', 'agenda', 'calendar', 'mail', 'gantt'];
            tabs.forEach(t => {
                document.getElementById(`tab-${t}`).classList.add('hidden');
                document.getElementById(`btn-${t}`).classList.remove('active');
            });
            document.getElementById(`tab-${tabName}`).classList.remove('hidden');
            document.getElementById(`btn-${tabName}`).classList.add('active');
        }

        // Función de Sincronización Cloud simulada
        function syncWithCloud() {
            const btn = document.getElementById('sync-btn');
            const status = document.getElementById('sync-status');
            const statusDot = document.getElementById('status-dot');
            const syncMsg = document.getElementById('sync-message');
            
            btn.innerText = "SINCRONIZANDO...";
            status.innerText = "ACTUALIZANDO...";
            status.className = "text-neonOrange font-bold syncing";
            statusDot.className = "w-3 h-3 bg-neonOrange rounded-full animate-ping";
            syncMsg.innerText = "Actualizando datos de correo, Google Calendar y Carta Gantt...";

                btn.innerText = "SINCRONIZAR COMPONENTES";
